-- ── Enum ─────────────────────────────────────────────────────────────────────

-- ── EDEK store (core HSM service) ────────────────────────────────────────────
CREATE TABLE hsm.edek_records (
    edek_id             UUID            PRIMARY KEY DEFAULT gen_random_uuid(),
    app_id              VARCHAR(128)    NOT NULL,
    edek_blob           TEXT            NOT NULL,          -- base64 RSA-wrapped DEK
    kek_version         VARCHAR(64)     NOT NULL,
    algorithm           VARCHAR(32)     NOT NULL DEFAULT 'AES-256-GCM',
    encoding            VARCHAR(16)     NOT NULL DEFAULT 'utf8',
    data_classification VARCHAR(32)     NULL,
    rotation_status     hsm.rotation_status NOT NULL DEFAULT 'current',
    created_at          TIMESTAMPTZ     NOT NULL DEFAULT now(),
    rotated_at          TIMESTAMPTZ     NULL,
    fingerprint         VARCHAR(16)     NULL       -- SHA-256(iv||tag)[0:8] as hex; NULL = pre-migration row, skip check
);

CREATE INDEX idx_edek_app_id          ON hsm.edek_records (app_id);
CREATE INDEX idx_edek_rotation_status ON hsm.edek_records (rotation_status);
CREATE INDEX idx_edek_kek_version     ON hsm.edek_records (kek_version);
CREATE INDEX idx_edek_classification  ON hsm.edek_records (data_classification);
CREATE INDEX idx_edek_created_at      ON hsm.edek_records (created_at);

-- ── App registry ─────────────────────────────────────────────────────────────
CREATE TABLE hsm.app_registrations (
    app_id          VARCHAR(128)    PRIMARY KEY,
    allowed_scopes  VARCHAR(512)    NOT NULL,   -- comma-separated: "encrypt,decrypt"
    description     VARCHAR(512)    NOT NULL DEFAULT '',
    active          BOOLEAN         NOT NULL DEFAULT TRUE
);

-- ── Cross-app decrypt grants (default-deny; own app always allowed) ───────────

    grantee_app_id  VARCHAR(128)    NOT NULL,
    owner_app_id    VARCHAR(128)    NOT NULL,
    PRIMARY KEY (grantee_app_id, owner_app_id)
);

    app_id          VARCHAR(128)    NOT NULL,
    edek_blob       TEXT            NOT NULL,
    kek_version     VARCHAR(64)     NOT NULL,

);

CREATE INDEX idx_edek_rotation_status  ON edek_records (rotation_status);
CREATE INDEX idx_edek_kek_version      ON edek_records (kek_version);

CREATE INDEX idx_edek_created_at       ON edek_records (created_at);

);
