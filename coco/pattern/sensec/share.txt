3 tables must exist before the pod starts in production. Here is the DDL — run this against your Postgres database once:

-- 1. EDEK Records — stores wrapped DEKs
CREATE TYPE rotation_status AS ENUM ('current', 'pending', 'rotated');

CREATE TABLE edek_records (
    edek_id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    app_id          VARCHAR(128)    NOT NULL,
    edek_blob       TEXT            NOT NULL,
    kek_version     VARCHAR(64)     NOT NULL,
    algorithm       VARCHAR(32)     NOT NULL DEFAULT 'AES-256-GCM',
    encoding        VARCHAR(16)     NOT NULL DEFAULT 'utf8',
    data_classification VARCHAR(32) NULL,
    rotation_status rotation_status NOT NULL DEFAULT 'current',
    created_at      TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
    rotated_at      TIMESTAMPTZ     NULL
);

CREATE INDEX idx_edek_app_id           ON edek_records (app_id);
CREATE INDEX idx_edek_rotation_status  ON edek_records (rotation_status);
CREATE INDEX idx_edek_kek_version      ON edek_records (kek_version);
CREATE INDEX idx_edek_classification   ON edek_records (data_classification);
CREATE INDEX idx_edek_created_at       ON edek_records (created_at);

-- 2. App Registrations — which apps can encrypt/decrypt
CREATE TABLE app_registrations (
    app_id          VARCHAR(128)    PRIMARY KEY,
    allowed_scopes  VARCHAR(512)    NOT NULL,   -- comma-separated: encrypt,decrypt,rotate
    description     VARCHAR(512)    NOT NULL DEFAULT '',
    active          BOOLEAN         NOT NULL DEFAULT TRUE
);

-- 3. App Decrypt Grants — cross-app decrypt permissions
CREATE TABLE app_decrypt_grants (
    grantee_app_id  VARCHAR(128)    NOT NULL,
    owner_app_id    VARCHAR(128)    NOT NULL,
    PRIMARY KEY (grantee_app_id, owner_app_id)
);

-- Seed at least one app registration so the service can accept requests
INSERT INTO app_registrations (app_id, allowed_scopes, description, active)
VALUES ('your-app-id', 'encrypt,decrypt', 'Production app', TRUE);
Important: the app_registrations table must have at least one row seeded before any API call works — without it every request fails with Unknown or inactive app_id. In demo mode this seeding happens automatically; in production you do it manually via the INSERT above or via the admin API after startup.
